cmake_minimum_required(VERSION 3.24)
project(StarterProject VERSION 0.1.0 LANGUAGES CXX)
if(NOT TARGET Dokime::Cpp)
  if(DOKIME_SDK_URL)
    include(FetchContent)
    FetchContent_Declare(dokime_sdk URL "${DOKIME_SDK_URL}"
      URL_HASH "SHA256=${DOKIME_SDK_SHA256}")
    FetchContent_MakeAvailable(dokime_sdk)
    list(PREPEND CMAKE_PREFIX_PATH "${dokime_sdk_SOURCE_DIR}")
  endif()
  find_package(Dokime CONFIG REQUIRED)
endif()
add_library(starter-project STATIC src/plugin.cpp src/plugin/operations/echo.cpp
  src/plugin/suites/capture/capture.cpp src/plugin/suites/capture/confirmation.cpp)
target_link_libraries(starter-project PUBLIC Dokime::Cpp)
add_executable(dokime-plugin src/plugin/main.cpp)
target_link_libraries(dokime-plugin PRIVATE starter-project)
set_target_properties(dokime-plugin PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/assets/plugin")
add_executable(dokime-client src/client/main.cpp)
target_link_libraries(dokime-client PRIVATE starter-project)
