cmake_minimum_required(VERSION 3.24)
# Without a Bundled SDK Package, Build the SDK and Its Native Dependencies From Source Through vcpkg.
if(NOT DOKIME_SDK_URL AND NOT CMAKE_PREFIX_PATH AND NOT CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VCPKG_ROOT})
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake")
endif()
# Native Dependencies Build Once: Release Only, Static Libraries on the Dynamic Release Runtime.
if(CMAKE_HOST_WIN32)
  set(VCPKG_TARGET_TRIPLET x64-windows-static-md-release CACHE STRING "Dependency Triplet")
else()
  set(VCPKG_TARGET_TRIPLET x64-linux-release CACHE STRING "Dependency Triplet")
endif()
set(VCPKG_HOST_TRIPLET "${VCPKG_TARGET_TRIPLET}" CACHE STRING "Code Generator Triplet")
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(StarterProject VERSION 0.1.0 LANGUAGES CXX)
if(NOT TARGET Dokime::Plugin)
  if(DOKIME_SDK_URL)
    include(FetchContent)
    FetchContent_Declare(dokime_sdk URL "${DOKIME_SDK_URL}"
      URL_HASH "SHA256=${DOKIME_SDK_SHA256}")
    FetchContent_MakeAvailable(dokime_sdk)
    list(PREPEND CMAKE_PREFIX_PATH "${dokime_sdk_SOURCE_DIR}")
  endif()
  find_package(Dokime CONFIG REQUIRED)
endif()
add_library(starter-project STATIC src/plugin.cpp src/plugin/operations/echo.cpp
  src/plugin/suites/capture/capture.cpp src/plugin/suites/capture/confirmation.cpp)
target_link_libraries(starter-project PUBLIC Dokime::Plugin)
add_executable(dokime-plugin src/plugin/main.cpp)
target_link_libraries(dokime-plugin PRIVATE starter-project Dokime::Client)
set_target_properties(dokime-plugin PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/assets/plugin")
add_executable(dokime-client src/client/main.cpp)
target_link_libraries(dokime-client PRIVATE starter-project Dokime::Client)
